### Category 1: Trait Contract Validation (UNIT)

Tests that verify the `FileHandler` trait compiles and behaves correctly.

| ID | Priority | Test | Expected Behavior |
|----|----------|------|-------------------|
| 5.1-U-001 | P0 | Trait with async methods compiles | `#[async_trait]` macro works correctly |

| 5.1-U-003 | P0 | Default implementations return Ok(None) | All 7 async methods decline by default |

```rust
#[tokio::test]
async fn test_async_trait_implementation() {
```

### Category 3: Handler Result Semantics (INT)

Tests for Ok(Some), Ok(None), Err chaining behavior.

| ID | Priority | Test | Expected Behavior |
|----|----------|------|-------------------|
| 5.1-I-004 | P0 | Ok(Some(data)) stops chain | First handler that accepts wins |

| 5.1-I-008 | P1 | Err from middle handler stops chain | Error position doesn't matter |

#### Test Implementation: 5.1-I-004

```rust
#[tokio::test]
async fn test_ok_some_stops_chain() {
    struct FirstAccepts;
    struct SecondNeverCalled;

    static SECOND_CALLED: AtomicBool = AtomicBool::new(false);

    #[async_trait]
    impl FileHandler for FirstAccepts {
        fn name(&self) -> &str { "first" }
        fn priority(&self) -> u32 { 10 }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            Ok(Some(b"first_data".to_vec()))
        }
    }

    #[async_trait]
    impl FileHandler for SecondNeverCalled {
        fn name(&self) -> &str { "second" }
        fn priority(&self) -> u32 { 20 }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            SECOND_CALLED.store(true, Ordering::SeqCst);
            Ok(Some(b"second_data".to_vec()))
        }
    }

    let fs = Arc::new(MockFileSystem::new());
    let mut registry = HandlerRegistry::with_filesystem(fs);
    registry.register(Arc::new(FirstAccepts));
    registry.register(Arc::new(SecondNeverCalled));

    let result = registry.handle_read("/test", 0, 100).await.unwrap();
    assert_eq!(result, b"first_data");
    assert!(!SECOND_CALLED.load(Ordering::SeqCst));
}
```

### Category 4: Read Operation (UNIT + INT)

| ID | Priority | Test | Expected Behavior |
|----|----------|------|-------------------|
| 5.1-U-010 | P0 | Read with offset/size slices correctly | Byte range extraction |
| 5.1-U-011 | P1 | Read with offset >= content returns empty | EOF handling |
| 5.1-U-012 | P1 | Read with size > remaining returns partial | Partial read |
| 5.1-U-013 | P2 | Zero-size read returns empty vec | Edge case |
| 5.1-I-009 | P1 | Read delegates to handler that can_handle | Path matching |

#### Test Implementation: 5.1-U-010

```rust
#[tokio::test]
async fn test_read_offset_size_slicing() {
    struct ContentHandler;

    #[async_trait]
    impl FileHandler for ContentHandler {
        fn name(&self) -> &str { "content" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, offset: u64, size: u64) -> HandlerResult<Vec<u8>> {
            let content = b"Hello, World!"; // 13 bytes
            let start = offset as usize;
            let end = (offset + size) as usize;

            if start >= content.len() {
                return Ok(Some(vec![]));
            }
            Ok(Some(content[start..end.min(content.len())].to_vec()))
        }
    }

    let h = ContentHandler;

    // Read from middle
    let result = h.read("/test", 7, 5).await.unwrap().unwrap();
    assert_eq!(result, b"World");

    // Read beyond end
    let result = h.read("/test", 7, 100).await.unwrap().unwrap();
    assert_eq!(result, b"World!");

    // Read from offset beyond content
    let result = h.read("/test", 100, 10).await.unwrap().unwrap();
    assert!(result.is_empty());
}
```

| ID | Priority | Test | Expected Behavior |

| 5.1-U-015 | P1 | Write at offset appends correctly | Offset handling |

| 5.1-I-010 | P1 | Write delegates to matching handler | Path-based routing |

#### Test Implementation: 5.1-U-014

```rust
#[tokio::test]
async fn test_write_returns_bytes_written() {
    struct WriteHandler {
        written: Arc<Mutex<Vec<u8>>>,
    }

    #[async_trait]
    impl FileHandler for WriteHandler {
        fn name(&self) -> &str { "write" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn write(&self, _: &str, offset: u64, data: &[u8]) -> HandlerResult<usize> {
            let mut buf = self.written.lock().unwrap();
            // Extend buffer if needed
            if buf.len() < offset as usize + data.len() {
                buf.resize(offset as usize + data.len(), 0);
            }
            buf[offset as usize..offset as usize + data.len()].copy_from_slice(data);
            Ok(Some(data.len()))
        }
    }

    let handler = WriteHandler { written: Arc::new(Mutex::new(vec![])) };

    let result = handler.write("/test", 0, b"hello").await.unwrap().unwrap();
    assert_eq!(result, 5);

    let result = handler.write("/test", 5, b" world").await.unwrap().unwrap();
    assert_eq!(result, 6);
}
```

```rust
#[tokio::test]
async fn test_concurrent_reads_no_race() {
    use std::sync::atomic::{AtomicUsize, Ordering};

    struct ConcurrentHandler {
        call_count: AtomicUsize,
    }

    #[async_trait]
    impl FileHandler for ConcurrentHandler {
        fn name(&self) -> &str { "concurrent" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            self.call_count.fetch_add(1, Ordering::SeqCst);
            tokio::time::sleep(Duration::from_millis(10)).await;
            Ok(Some(b"data".to_vec()))
        }
    }

    let handler = Arc::new(ConcurrentHandler { call_count: AtomicUsize::new(0) });
    let fs = Arc::new(MockFileSystem::new());
    let mut registry = HandlerRegistry::with_filesystem(fs);
    registry.register(handler.clone());

    // Launch 10 concurrent reads
    let mut handles = vec![];
    for i in 0..10 {
        let reg = registry.clone(); // Requires Clone impl
        handles.push(tokio::spawn(async move {
            reg.handle_read(&format!("/file{}", i), 0, 100).await
        }));
    }

    // Wait for all to complete
    for h in handles {
        assert!(h.await.unwrap().is_ok());
    }

    assert_eq!(handler.call_count.load(Ordering::SeqCst), 10);
}
```
