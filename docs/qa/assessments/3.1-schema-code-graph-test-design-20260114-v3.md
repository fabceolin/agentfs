### TC-3.1-01: Symbol insertion with valid inode

```gherkin
GIVEN fs_current contains inode 100 for file 'src/main.rs'
WHEN inserting code_symbol with:
  - id = 'src/main.rs:main'
  - inode = 100
  - name = 'main'
  - kind = 'function'
  - language = 'rust'
THEN symbol is created successfully
AND symbol is queryable by id, name, kind, and language indexes
```

**SQL Implementation:**
```sql
-- Setup: Create file in fs_journal
INSERT INTO fs_journal (inode, event_type, parent, name, mode)
VALUES (100, 'create', 1, 'main.rs', 33188);

-- Test: Insert symbol
INSERT INTO code_symbols (id, inode, name, kind, language)
VALUES ('src/main.rs:main', 100, 'main', 'function', 'rust');

-- Verify creation
SELECT COUNT(*) = 1 AS test_passed FROM code_symbols
WHERE id = 'src/main.rs:main';

-- Verify index lookups work
SELECT COUNT(*) = 1 AS name_index_works FROM code_symbols WHERE name = 'main';
SELECT COUNT(*) = 1 AS kind_index_works FROM code_symbols WHERE kind = 'function';
SELECT COUNT(*) = 1 AS language_index_works FROM code_symbols WHERE language = 'rust';
```

### TC-3.1-03: Dependency cascade on symbol deletion

```gherkin
GIVEN code_symbol 'test:a' exists with inode 200
AND code_symbol 'test:b' exists with inode 200
AND code_dependency 'test:a' -> 'test:b' (calls) exists
WHEN deleting code_symbol 'test:a'
THEN code_dependencies with source_id = 'test:a' are cascade deleted
AND code_symbol 'test:b' remains unaffected
```

**SQL Implementation:**
```sql
-- Setup
INSERT INTO fs_journal (inode, event_type, parent, name, mode)
VALUES (200, 'create', 1, 'test.rs', 33188);

INSERT INTO code_symbols (id, inode, name, kind)
VALUES ('test:a', 200, 'a', 'function'),
       ('test:b', 200, 'b', 'function');

INSERT INTO code_dependencies (source_id, target_id, dep_type)
VALUES ('test:a', 'test:b', 'calls');

-- Verify setup
SELECT COUNT(*) = 1 AS dep_exists FROM code_dependencies
WHERE source_id = 'test:a';

-- Delete symbol (should cascade)
DELETE FROM code_symbols WHERE id = 'test:a';

-- Verify cascade
SELECT COUNT(*) = 0 AS dep_cascaded FROM code_dependencies
WHERE source_id = 'test:a';

-- Verify test:b unaffected
SELECT COUNT(*) = 1 AS target_preserved FROM code_symbols
WHERE id = 'test:b';
```

### TC-3.1-04: Transitive dependency query with cycles

```gherkin
GIVEN symbols A, B, C exist (all with inode 300)
AND dependencies: A -> B, B -> C, C -> A (cycle)
WHEN querying transitive dependencies from A with depth limit 10
THEN query terminates without infinite loop
AND returns B (depth 1), C (depth 2), A (depth 3)
AND execution time < 1 second
```

**SQL Implementation:**
```sql
-- Setup cyclic graph
INSERT INTO fs_journal (inode, event_type, parent, name, mode)
VALUES (300, 'create', 1, 'cycle.rs', 33188);

INSERT INTO code_symbols (id, inode, name, kind)
VALUES ('cycle:a', 300, 'a', 'function'),
       ('cycle:b', 300, 'b', 'function'),
       ('cycle:c', 300, 'c', 'function');

INSERT INTO code_dependencies (source_id, target_id, dep_type)
VALUES ('cycle:a', 'cycle:b', 'calls'),
       ('cycle:b', 'cycle:c', 'calls'),
       ('cycle:c', 'cycle:a', 'calls');  -- Creates cycle

-- Transitive closure with depth limit
WITH RECURSIVE deps AS (
    -- Base case
    SELECT target_id, 1 as depth
    FROM code_dependencies
    WHERE source_id = 'cycle:a'

    UNION ALL

    -- Recursive case with depth limit
    SELECT d.target_id, deps.depth + 1
    FROM code_dependencies d
    JOIN deps ON d.source_id = deps.target_id
    WHERE deps.depth < 10
)
SELECT DISTINCT target_id, MIN(depth) as min_depth
FROM deps
GROUP BY target_id
ORDER BY min_depth;

-- Expected results:
-- cycle:b | 1
-- cycle:c | 2
-- cycle:a | 3
-- Query should terminate (not infinite loop)
```

### TC-3.1-07: Impact analysis query (reverse transitive closure)

```gherkin
GIVEN symbols: main, process_data, Data, Output, process
AND dependency graph:
  - main -> process_data (calls)
  - process_data -> Data (uses)
  - process_data -> Output (uses)
  - process_data -> process (calls)
WHEN querying "what breaks if I change Data?"
THEN returns: process_data (distance 1), main (distance 2)
```

**SQL Implementation:**
```sql
-- Setup (using minimal dataset from story)
INSERT INTO fs_journal (inode, event_type, parent, name, mode)
VALUES (100, 'create', 1, 'main.rs', 33188),
       (101, 'create', 1, 'lib.rs', 33188);

INSERT INTO code_symbols (id, inode, name, kind, language)
VALUES
    ('src/main.rs:main', 100, 'main', 'function', 'rust'),
    ('src/main.rs:process_data', 100, 'process_data', 'function', 'rust'),
    ('src/lib.rs:Data', 101, 'Data', 'struct', 'rust'),
    ('src/lib.rs:Output', 101, 'Output', 'struct', 'rust'),
    ('src/lib.rs:process', 101, 'process', 'function', 'rust');

INSERT INTO code_dependencies (source_id, target_id, dep_type, weight)
VALUES
    ('src/main.rs:main', 'src/main.rs:process_data', 'calls', 1.0),
    ('src/main.rs:process_data', 'src/lib.rs:Data', 'uses', 0.5),
    ('src/main.rs:process_data', 'src/lib.rs:Output', 'uses', 0.5),
    ('src/main.rs:process_data', 'src/lib.rs:process', 'calls', 1.0);

-- Impact analysis: What depends on Data?
WITH RECURSIVE impact AS (
    -- Base case: direct dependents
    SELECT source_id, 1 as distance
    FROM code_dependencies
    WHERE target_id = 'src/lib.rs:Data'

    UNION ALL

    -- Recursive case: dependents of dependents
    SELECT d.source_id, impact.distance + 1
    FROM code_dependencies d
    JOIN impact ON d.target_id = impact.source_id
    WHERE impact.distance < 10
)
SELECT DISTINCT s.name, s.kind, MIN(i.distance) as impact_distance
FROM impact i
JOIN code_symbols s ON s.id = i.source_id
GROUP BY s.id, s.name, s.kind
ORDER BY impact_distance;

-- Expected:
-- process_data | function | 1
-- main         | function | 2
```

### TC-3.1-08: Dead code detection (orphan symbols)

```gherkin
GIVEN symbols: main, process_data, unused_function
AND dependencies: main -> process_data
AND unused_function has NO incoming edges
WHEN querying for orphan symbols (symbols with no callers)
THEN returns: unused_function
AND main is excluded (entry point)
```

**SQL Implementation:**
```sql
-- Setup
INSERT INTO fs_journal (inode, event_type, parent, name, mode)
VALUES (500, 'create', 1, 'orphan.rs', 33188);

INSERT INTO code_symbols (id, inode, name, kind)
VALUES ('orphan:unused', 500, 'unused_function', 'function');

-- No dependencies pointing to unused_function

-- Find dead code (symbols never called)
SELECT s.id, s.name, s.kind
FROM code_symbols s
WHERE s.kind IN ('function', 'method')
  AND s.name != 'main'  -- Exclude entry points
  AND NOT EXISTS (
      SELECT 1 FROM code_dependencies d
      WHERE d.target_id = s.id
        AND d.dep_type = 'calls'
  );

-- Expected: orphan:unused | unused_function | function
```

## DuckDB-Specific Implementation Notes

### Foreign Key Constraints

DuckDB enforces FK constraints. Expected error format:
```
Constraint Error: Violates foreign key constraint because key "inode: 999"
does not exist in the referenced table "fs_current"
```

### Index Introspection

```sql
SELECT index_name, table_name, is_unique
FROM duckdb_indexes()
WHERE table_name IN ('code_symbols', 'code_dependencies');
```

### DuckPGQ Extension Check

```sql
-- Before running property graph tests
SELECT installed, loaded
FROM duckdb_extensions()
WHERE extension_name = 'duckpgq';

-- Skip tests if not available
```

### Cascade Delete Behavior

DuckDB supports `ON DELETE CASCADE` but requires explicit definition:
```sql
FOREIGN KEY (source_id) REFERENCES code_symbols(id) ON DELETE CASCADE
```

**Note:** Current schema may need modification to enable cascade deletes on code_dependencies.
