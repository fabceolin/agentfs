**5.1-INT-005: Dispatch Respects Priority**
```rust
#[tokio::test]
async fn test_high_priority_tried_first() {
    static CALL_ORDER: Mutex<Vec<&str>> = Mutex::new(Vec::new());

    struct TrackingHandler { name: &'static str, priority: u32 }

    #[async_trait]
    impl FileHandler for TrackingHandler {
        fn name(&self) -> &str { self.name }
        fn priority(&self) -> u32 { self.priority }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            CALL_ORDER.lock().unwrap().push(self.name);
            Ok(None) // Decline to allow next handler
        }
    }

    let mut registry = HandlerRegistry::with_filesystem(Arc::new(MockFileSystem::new()));
    registry.register(Arc::new(TrackingHandler { name: "low", priority: 100 }));
    registry.register(Arc::new(TrackingHandler { name: "high", priority: 10 }));

    let _ = registry.handle_read("/test", 0, 100).await;

    let order = CALL_ORDER.lock().unwrap();
    assert_eq!(order[0], "high");
    assert_eq!(order[1], "low");
}
```

**5.1-UNIT-012: Read Offset/Size Slicing**
```rust
#[tokio::test]
async fn test_read_slicing() {
    struct ContentHandler;

    #[async_trait]
    impl FileHandler for ContentHandler {
        fn name(&self) -> &str { "content" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, offset: u64, size: u64) -> HandlerResult<Vec<u8>> {
            let content = b"Hello, World!"; // 13 bytes
            let start = offset as usize;
            let end = (offset + size) as usize;

            if start >= content.len() {
                return Ok(Some(vec![]));
            }
            Ok(Some(content[start..end.min(content.len())].to_vec()))
        }
    }

    let h = ContentHandler;

    // Full read
    assert_eq!(h.read("/test", 0, 100).await.unwrap().unwrap(), b"Hello, World!");

    // Partial read from middle
    assert_eq!(h.read("/test", 7, 5).await.unwrap().unwrap(), b"World");

    // Read past end
    assert_eq!(h.read("/test", 7, 100).await.unwrap().unwrap(), b"World!");

    // Read from beyond content
    assert_eq!(h.read("/test", 100, 10).await.unwrap().unwrap(), b"");
}
```

**5.1-UNIT-016: Write Returns Bytes Written**
```rust
#[tokio::test]
async fn test_write_returns_count() {
    struct WriteHandler {
        buffer: Arc<Mutex<Vec<u8>>>,
    }

    #[async_trait]
    impl FileHandler for WriteHandler {
        fn name(&self) -> &str { "write" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn write(&self, _: &str, offset: u64, data: &[u8]) -> HandlerResult<usize> {
            let mut buf = self.buffer.lock().unwrap();
            let end = offset as usize + data.len();
            if buf.len() < end {
                buf.resize(end, 0);
            }
            buf[offset as usize..end].copy_from_slice(data);
            Ok(Some(data.len()))
        }
    }

    let handler = WriteHandler { buffer: Arc::new(Mutex::new(vec![])) };

    assert_eq!(handler.write("/test", 0, b"hello").await.unwrap().unwrap(), 5);
    assert_eq!(handler.write("/test", 5, b" world").await.unwrap().unwrap(), 6);

    let buf = handler.buffer.lock().unwrap();
    assert_eq!(&*buf, b"hello world");
}
```

### AC4: Handler Result Type Semantics

**Requirement:** Ok(Some), Ok(None), Err have specific chaining behaviors.

| ID | Level | Priority | Test | Justification |
|----|-------|----------|------|---------------|
| 5.1-INT-017 | Integration | P0 | Ok(Some(data)) stops chain and returns data | First accepting handler wins |
| 5.1-INT-018 | Integration | P0 | Ok(None) passes to next handler in chain | Decline semantics |
| 5.1-INT-019 | Integration | P0 | Err stops chain with error (no further handlers tried) | Error propagation |
| 5.1-INT-020 | Integration | P1 | All handlers decline = default handler used | Fallback behavior |
| 5.1-INT-021 | Integration | P1 | Err from middle handler stops chain | Error position independence |
| 5.1-E2E-001 | E2E | P1 | Error from handler appears as FUSE errno | Error translation |
| 5.1-E2E-002 | E2E | P2 | Full chain: high-priority declines, low-priority handles | Complete flow |
| 5.1-E2E-003 | E2E | P3 | Handler error triggers trace log | Observability |

#### Detailed Scenarios

**5.1-INT-017: Ok(Some) Stops Chain**
```rust
#[tokio::test]
async fn test_ok_some_stops_chain() {
    static SECOND_CALLED: AtomicBool = AtomicBool::new(false);

    struct FirstAccepts;
    struct SecondNeverCalled;

    #[async_trait]
    impl FileHandler for FirstAccepts {
        fn name(&self) -> &str { "first" }
        fn priority(&self) -> u32 { 10 }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            Ok(Some(b"first_data".to_vec()))
        }
    }

    #[async_trait]
    impl FileHandler for SecondNeverCalled {
        fn name(&self) -> &str { "second" }
        fn priority(&self) -> u32 { 20 }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            SECOND_CALLED.store(true, Ordering::SeqCst);
            Ok(Some(b"second_data".to_vec()))
        }
    }

    let mut registry = HandlerRegistry::with_filesystem(Arc::new(MockFileSystem::new()));
    registry.register(Arc::new(FirstAccepts));
    registry.register(Arc::new(SecondNeverCalled));

    let result = registry.handle_read("/test", 0, 100).await.unwrap();

    assert_eq!(result, b"first_data");
    assert!(!SECOND_CALLED.load(Ordering::SeqCst), "Second handler should not be called");
}
```

**5.1-INT-022: Concurrent Reads No Race**
```rust
#[tokio::test]
async fn test_concurrent_reads() {
    use std::sync::atomic::AtomicUsize;

    struct ConcurrentHandler {
        call_count: AtomicUsize,
    }

    #[async_trait]
    impl FileHandler for ConcurrentHandler {
        fn name(&self) -> &str { "concurrent" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            self.call_count.fetch_add(1, Ordering::SeqCst);
            tokio::time::sleep(Duration::from_millis(10)).await;
            Ok(Some(b"data".to_vec()))
        }
    }

    let handler = Arc::new(ConcurrentHandler { call_count: AtomicUsize::new(0) });
    let mut registry = HandlerRegistry::with_filesystem(Arc::new(MockFileSystem::new()));
    registry.register(handler.clone());

    let mut handles = vec![];
    for i in 0..10 {
        let reg = registry.clone();
        handles.push(tokio::spawn(async move {
            reg.handle_read(&format!("/file{}", i), 0, 100).await
        }));
    }

    for h in handles {
        assert!(h.await.unwrap().is_ok());
    }

    assert_eq!(handler.call_count.load(Ordering::SeqCst), 10);
}
```
