```rust
#[tokio::test]
async fn test_dispatch_respects_priority_order() {
    use std::sync::atomic::{AtomicUsize, Ordering};

    static CALL_ORDER: AtomicUsize = AtomicUsize::new(0);
    static FIRST_ORDER: AtomicUsize = AtomicUsize::new(0);
    static SECOND_ORDER: AtomicUsize = AtomicUsize::new(0);

    struct HighPriority;
    struct LowPriority;

    #[async_trait]
    impl FileHandler for HighPriority {
        fn name(&self) -> &str { "high" }
        fn priority(&self) -> u32 { 10 }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            FIRST_ORDER.store(CALL_ORDER.fetch_add(1, Ordering::SeqCst), Ordering::SeqCst);
            Ok(None) // Decline
        }
    }

    #[async_trait]
    impl FileHandler for LowPriority {
        fn name(&self) -> &str { "low" }
        fn priority(&self) -> u32 { 100 }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            SECOND_ORDER.store(CALL_ORDER.fetch_add(1, Ordering::SeqCst), Ordering::SeqCst);
            Ok(Some(b"data".to_vec()))
        }
    }

    let fs = Arc::new(MockFileSystem::new());
    let mut registry = HandlerRegistry::with_filesystem(fs);
    registry.register(Arc::new(LowPriority));  // Register low first
    registry.register(Arc::new(HighPriority)); // But high should be called first

    let _ = registry.handle_read("/test", 0, 100).await;

    assert!(FIRST_ORDER.load(Ordering::SeqCst) < SECOND_ORDER.load(Ordering::SeqCst),
        "High priority handler should be called before low priority");
}
```

```rust
#[tokio::test]
async fn test_read_offset_slice() {
    struct ContentHandler;

    #[async_trait]
    impl FileHandler for ContentHandler {
        fn name(&self) -> &str { "content" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, _: &str, offset: u64, size: u64) -> HandlerResult<Vec<u8>> {
            let content = b"Hello, World!"; // 13 bytes: indices 0-12
            let start = offset as usize;
            let end = (offset + size) as usize;

            if start >= content.len() {
                return Ok(Some(vec![]));
            }
            Ok(Some(content[start..end.min(content.len())].to_vec()))
        }
    }

    let h = ContentHandler;

    // Full content
    assert_eq!(h.read("/test", 0, 100).await.unwrap().unwrap(), b"Hello, World!");

    // Middle slice: "World" starts at index 7
    assert_eq!(h.read("/test", 7, 5).await.unwrap().unwrap(), b"World");

    // Slice extending past end
    assert_eq!(h.read("/test", 7, 100).await.unwrap().unwrap(), b"World!");

    // Offset beyond content
    assert!(h.read("/test", 100, 10).await.unwrap().unwrap().is_empty());

    // Single byte at boundary
    assert_eq!(h.read("/test", 12, 1).await.unwrap().unwrap(), b"!");
}
```

```rust
#[tokio::test]
async fn test_write_returns_bytes_written() {
    use std::sync::Mutex;

    struct WriteHandler {
        buffer: Arc<Mutex<Vec<u8>>>,
    }

    #[async_trait]
    impl FileHandler for WriteHandler {
        fn name(&self) -> &str { "write" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn write(&self, _: &str, offset: u64, data: &[u8]) -> HandlerResult<usize> {
            let mut buf = self.buffer.lock().unwrap();
            let end = offset as usize + data.len();
            if buf.len() < end {
                buf.resize(end, 0);
            }
            buf[offset as usize..end].copy_from_slice(data);
            Ok(Some(data.len()))
        }
    }

    let handler = WriteHandler { buffer: Arc::new(Mutex::new(vec![])) };

    // First write
    let result = handler.write("/test", 0, b"hello").await.unwrap().unwrap();
    assert_eq!(result, 5);

    // Second write at offset
    let result = handler.write("/test", 5, b" world").await.unwrap().unwrap();
    assert_eq!(result, 6);

    // Verify combined content
    let buf = handler.buffer.lock().unwrap();
    assert_eq!(&*buf, b"hello world");
}
```

```rust
        fn name(&self) -> &str { "consistent" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }

        async fn getattr(&self, _: &str) -> HandlerResult<Stats> {

                ino: 42,
                mode: 0o100644,
                nlink: 1,
                uid: 1000,
                gid: 1000,
                size: CONTENT.len() as i64,
                atime: 0,
                mtime: 0,
                ctime: 0,
            }))
        }

        async fn read(&self, _: &str, offset: u64, size: u64) -> HandlerResult<Vec<u8>> {
            let start = offset as usize;
            let end = (offset + size) as usize;
            if start >= CONTENT.len() {
                return Ok(Some(vec![]));
            }
            Ok(Some(CONTENT[start..end.min(CONTENT.len())].to_vec()))
        }
    }

    let h = ConsistentHandler;

    let stats = h.getattr("/test").await.unwrap().unwrap();
    let content = h.read("/test", 0, 1000).await.unwrap().unwrap();

    assert_eq!(stats.size as usize, content.len(),
        "getattr size must match actual content length");
```

```rust
#[tokio::test]
async fn test_concurrent_reads_no_data_race() {
    use std::sync::atomic::{AtomicUsize, Ordering};

    struct ConcurrentHandler {
        call_count: AtomicUsize,
    }

    #[async_trait]
    impl FileHandler for ConcurrentHandler {
        fn name(&self) -> &str { "concurrent" }
        fn can_handle(&self, _: &str, _: Option<&Stats>) -> bool { true }
        async fn read(&self, path: &str, _: u64, _: u64) -> HandlerResult<Vec<u8>> {
            self.call_count.fetch_add(1, Ordering::SeqCst);
            // Simulate async work
            tokio::time::sleep(Duration::from_millis(10)).await;
            Ok(Some(format!("data for {}", path).into_bytes()))
        }
    }

    let handler = Arc::new(ConcurrentHandler { call_count: AtomicUsize::new(0) });
    let fs = Arc::new(MockFileSystem::new());
    let mut registry = HandlerRegistry::with_filesystem(fs);
    registry.register(handler.clone());

    // Launch 10 concurrent reads
    let mut handles = vec![];
    for i in 0..10 {
        let path = format!("/file{}", i);
        let reg = registry.clone();
        handles.push(tokio::spawn(async move {
            reg.handle_read(&path, 0, 100).await
        }));
    }

    // Wait for all to complete
    for h in handles {
        let result = h.await.unwrap();
        assert!(result.is_ok(), "Concurrent read should succeed");
    }

    assert_eq!(handler.call_count.load(Ordering::SeqCst), 10,
        "All 10 concurrent reads should complete");
}
```
